    <script>
        let currentLang = 'en';

        function toggleLanguage() {
            const langEnElements = document.querySelectorAll('.lang-en');
            const langJpElements = document.querySelectorAll('.lang-jp');
            const btnText = document.getElementById('lang-btn-text');

            if (currentLang === 'en') {
                langEnElements.forEach(el => el.classList.add('hidden'));
                langJpElements.forEach(el => el.classList.remove('hidden'));
                btnText.textContent = 'English';
                currentLang = 'jp';
            } else {
                langJpElements.forEach(el => el.classList.add('hidden'));
                langEnElements.forEach(el => el.classList.remove('hidden'));
                btnText.textContent = '日本語';
                currentLang = 'en';
            }
        }
    </script>
